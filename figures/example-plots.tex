\newcommand{\methoddiagram}{%
  \begin{tikzpicture}[
    stage/.style={draw=kitfziprimary,line width=1.5pt,rounded corners=5pt,
      fill=kitfzilight,text width=6.7cm,minimum height=3cm,align=center,font=\small},
    flow/.style={-{Latex[length=6mm]},line width=1.7pt}]
    \node[stage] (input) {\textbf{Input}\\$x$};
    \node[stage,right=1.6cm of input] (representation) {\textbf{Representation}\\$h=f_\theta(x)$};
    \node[stage,right=1.6cm of representation] (output) {\textbf{Prediction}\\$\hat y=g_\phi(h)$};
    \node[stage,below=2cm of representation,fill=white] (component) {\textbf{Proposed component}\\Your main idea};
    \draw[flow] (input) -- (representation);
    \draw[flow] (representation) -- (output);
    \draw[flow,kitfziprimary] (component) -- (representation);
    \draw[flow,kitfzisecondary,dashed] (input.north) -- ++(0,1.6cm) -| (output.north);
    \node[above=1.6cm of representation,font=\small,text=kitfzisecondary] {Baseline comparison};
  \end{tikzpicture}%
}

\newcommand{\learningplot}[3]{%
  \begin{tikzpicture}
    \begin{axis}[kitfzi axis,width=0.84\linewidth,height=8cm,
      xmin=0,xmax=50,ymin=0.35,ymax=1,xlabel={Epoch},ylabel={Score},
      title={#3},title style={font=\small},legend pos=south east,
      domain=0:50,samples=51]
      \addplot[kitfziprimary] {#2-0.48*exp(-x/(10+#1*12))};
      \addplot[kitfzisecondary,dashed] {#2-0.10-0.4*exp(-x/(13+#1*12))};
      \legend{Method,Baseline}
    \end{axis}
  \end{tikzpicture}%
}
