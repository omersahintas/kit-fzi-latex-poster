  \begin{block}{Research questions}
    \begin{enumerate}
      \item What problem matters, and who benefits from solving it?
      \item Which limitation of existing methods motivates your work?
      \item What evidence would demonstrate a meaningful improvement?
    \end{enumerate}
    Introduce the task and the setting in two or three sentences. State the
    assumptions that define the scope of your research.

  \begin{block}{Contributions}
    \begin{enumerate}
      \item \textbf{Idea.} Describe the main conceptual contribution in one sentence.
      \item \textbf{Reproducible method.} Explain what others can implement or reuse.
      \item \textbf{Evidence and limitations.} Summarize what the experiments establish.
    \end{enumerate}
  \end{block}

  \begin{block}{Method}
    \textbf{1. Input.} Specify the observations, labels, or signals available to the method.

    \textbf{2. Representation.} Introduce the core transformation and the proposed component.

      \hat y = g_\phi(h), \qquad h = f_\theta(x)
    \]
    Here, $x$ denotes an input, $h$ its representation, and $\hat y$ a prediction;
    $\theta$ and $\phi$ are learned parameters.

    \textbf{3. Evaluation.} Compare against appropriate baselines using the same protocol.
  \end{block}
